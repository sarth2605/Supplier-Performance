MySQL DDL Script (database/schema.sql)

-- Supplier Performance Analysis System DDL Script
CREATE DATABASE IF NOT EXISTS `supplier_performance_db`;
USE `supplier_performance_db`;

CREATE TABLE `suppliers` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `supplier_code` VARCHAR(30) NOT NULL UNIQUE,
    `supplier_name` VARCHAR(150) NOT NULL,
    `company_name` VARCHAR(150) NOT NULL,
    `email` VARCHAR(120) NOT NULL,
    `phone` VARCHAR(30) NOT NULL,
    `category` ENUM('Raw Materials', 'Electronics', 'Packaging', 'Logistics', 'IT Services', 'Machinery', 'Chemicals') NOT NULL,
    `contract_start_date` DATE NOT NULL,
    `contract_end_date` DATE NOT NULL,
    `status` ENUM('Active', 'Under Review', 'Inactive') DEFAULT 'Active'
) ENGINE=InnoDB;

CREATE TABLE `performance` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `supplier_id` INT NOT NULL,
    `evaluation_date` DATE NOT NULL,
    `quality_score` DECIMAL(5, 2) NOT NULL,
    `delivery_score` DECIMAL(5, 2) NOT NULL,
    `cost_score` DECIMAL(5, 2) NOT NULL,
    `reliability_score` DECIMAL(5, 2) NOT NULL,
    `service_score` DECIMAL(5, 2) NOT NULL,
    `defect_rate` DECIMAL(5, 2) NOT NULL DEFAULT 0.00,
    `overall_score` DECIMAL(5, 2) NOT NULL,
    `rating` ENUM('Excellent', 'Good', 'Average', 'Poor', 'Critical') NOT NULL,
    FOREIGN KEY (`supplier_id`) REFERENCES `suppliers`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB;

Project Viva & Defense Reference Notes

Section 26 Project Objective & Technical Defense Q&A

Official Viva Objective Statement:

“The main objective of the Supplier Performance Analysis System is to evaluate, monitor, compare and rank suppliers based on quality, delivery, cost, reliability and service performance. The system automatically calculates an overall performance score and provides dashboards, charts and reports to support better supplier selection and management decisions.”

Q1: How is the overall performance score computed?

Ans: It uses a weighted multi-attribute rating formula: Overall Score = (Quality × 0.30) + (Delivery × 0.25) + (Cost × 0.20) + (Reliability × 0.15) + (Service × 0.10).

Q2: What is the rating classification scale?

Ans: 90–100: Excellent, 80–89: Good, 70–79: Average, 60–69: Poor, Below 60: Critical.

Q3: How are risk alerts triggered?

Ans: The system automatically generates risk alerts when a supplier's score drops below 70%, defect rate exceeds 3.0%, or contract expires in under 45 days.
